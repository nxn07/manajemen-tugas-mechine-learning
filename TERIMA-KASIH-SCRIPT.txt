----------------------

# Option 1: Manual Commands (Recommended)
cd /mnt/c/Users/Microsoft/Documents/Ngoding/KP

# Stop all containers
podman-compose down

# Remove old containers
podman rm simkap_* 2>/dev/null || true

# Start fresh
podman-compose up -d

# Wait for health check
sleep 30

# Re-apply caches
podman exec simkap_backend php artisan config:cache
podman exec simkap_backend php artisan route:cache

# Check status
podman ps --format "{{.Names}} | {{.Status}}"

Troubleshooting:
---------------
If containers don't start:
  $ CMD logs simkap_backend --tail 50
  
If cache fails:
  $ CMD exec simkap_backend php artisan optimize:clear
  Then re-run: php artisan config:cache
  
If database issue:
  $ CMD exec simkap_backend php artisan db:connection --force
